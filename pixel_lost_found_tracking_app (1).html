เชื่อมต่อ Google Sheets

* เมื่อเชื่อมต่อแล้ว ข้อมูลการแจ้งของหายทั้งหมดจะดึงและบันทึกตรงเข้าตาราง Google Sheet ของคุณทันที!

วิธีตั้งค่า Google Sheet ใน 2 นาที:

  1. เปิด Google Sheets สร้างสเปรดชีตใหม่
  2. ไปที่เมนู ส่วนขยาย (Extensions) ➔ Apps Script
  3. ลบโค้ดเดิมออกทั้งหมด แล้วนำโค้ดด้านล่างนี้ไปวางแทนที่:
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var result = [];
  if (data.length > 1) {
    var headers = data[0];
    for (var i = 1; i < data.length; i++) {
      var obj = {};
      for (var j = 0; j < headers.length; j++) {
        obj[headers[j]] = data[i][j];
      }
      result.push(obj);
    }
  }
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  // Create headers if empty
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["id", "type", "title", "category", "location", "date", "contact", "reward", "status", "description", "imageUrl", "createdAt"]);
  }
  var data = JSON.parse(e.postData.contents);
  if (data.action === 'add') {
    sheet.appendRow([data.id, data.type, data.title, data.category, data.location, data.date, data.contact, data.reward, data.status, data.description, data.imageUrl, data.createdAt]);
    return ContentService.createTextOutput(JSON.stringify({status: 'success'})).setMimeType(ContentService.MimeType.JSON);
  } else if (data.action === 'updateStatus') {
    var values = sheet.getDataRange().getValues();
    for (var i = 1; i < values.length; i++) {
      if (values[i][0] == data.id) {
        sheet.getRange(i + 1, 9).setValue(data.status); // Column 9 = Status
        break;
      }
    }
    return ContentService.createTextOutput(JSON.stringify({status: 'success'})).setMimeType(ContentService.MimeType.JSON);
  }
}
  1. กดปุ่ม ทำให้ใช้งานได้ (Deploy) ➔ การทำให้ใช้งานได้ใหม่ (New deployment)
  2. เลือกประเภทเป็น เว็บแอป (Web app)
  3. ตั้งค่า ผู้มีสิทธิ์เข้าถึง (Who has access) เป็น ทุกคน (Anyone)
  4. กด Deploy ก๊อปปี้ Web App URL มาวางในช่องด้านบนแล้วกด "เชื่อมต่อ" ได้เลย!
